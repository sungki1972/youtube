## 6단계: PM2로 봇 실행
## ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

# CMD에서 mp3bot 폴더로 이동:
cd "C:\Users\neola\OneDrive\문서\cursor\25\8\mp3\mp3bot"

# logs 폴더 생성:
mkdir logs

# ecosystem.config.js 사용하여 실행:
pm2 start ecosystem.config.js

# 또는 직접 실행:
pm2 start telegram_mp3_bot.py --name mp3bot --interpreter python

# 상태 확인:
pm2 status

# 로그 보기:
pm2 logs mp3bot

# 서버 재부팅 후에도 자동 시작되도록 저장:
pm2 save
